-- 1. CREACIÓN Y USO DE LA BASE DE DATOS

CREATE DATABASE IF NOT EXISTS bd_keren_mass;
USE bd_keren_mass;


-- 2. CREACIÓN DE TABLAS (Ordenadas por dependencias)

CREATE TABLE usuario (
    id_usuario INT AUTO_INCREMENT PRIMARY KEY,
    usuario VARCHAR(50) NOT NULL UNIQUE,
    rol VARCHAR(30) NOT NULL,
    correo VARCHAR(100) NOT NULL UNIQUE,
    telefono VARCHAR(20),
    contrasena VARCHAR(255) NOT NULL,
    token INT,
    estado VARCHAR(20) NOT NULL DEFAULT 'CONTRATADO'
);

CREATE TABLE materia_prima (
    id_materia_prima VARCHAR(10) PRIMARY KEY,
    nombre_mp VARCHAR(100) NOT NULL,
    unidad_mp VARCHAR(30),
    stock_mp DECIMAL(10,2),
    precio_unitario DECIMAL(10,2),
    stock_minimo DECIMAL(10,2)
);

CREATE TABLE receta (
    id_receta VARCHAR(10) PRIMARY KEY,
    nom_receta VARCHAR(100),
    costo_produccion DECIMAL(10,2)
);

CREATE TABLE proveedor (
    id_proveedor VARCHAR(10) PRIMARY KEY,
    nombre_proveedor VARCHAR(100) NOT NULL,
    ruc VARCHAR(20),
    telefono VARCHAR(20),
    direccion VARCHAR(150),
    correo VARCHAR(100)
);


CREATE TABLE producto (
    id_producto VARCHAR(10) PRIMARY KEY,
    nom_producto VARCHAR(100),
    stock_prd INT,
    precio_prd DECIMAL(10,2),
    id_receta VARCHAR(10),
    stock_minimo INT,
    FOREIGN KEY (id_receta) REFERENCES receta(id_receta)
);

CREATE TABLE receta_mp (
    id_receta VARCHAR(10),
    id_materia_prima VARCHAR(10),
    cantidad DECIMAL(10,2),
    PRIMARY KEY (id_receta, id_materia_prima),
    FOREIGN KEY (id_receta) REFERENCES receta(id_receta),
    FOREIGN KEY (id_materia_prima) REFERENCES materia_prima(id_materia_prima)
);

CREATE TABLE cotizacion (
    id_cotizacion VARCHAR(10) NOT NULL,
    id_proveedor VARCHAR(10) NOT NULL,
    fecha_solicitud DATE NOT NULL,
    id_usuario INT NOT NULL,
    costo_total DECIMAL(10,2) DEFAULT 0.00,
    estado VARCHAR(30) DEFAULT 'ENVIADO',
    PRIMARY KEY (id_cotizacion, id_proveedor),
    FOREIGN KEY (id_proveedor) REFERENCES proveedor(id_proveedor),
    FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario)
);

CREATE TABLE venta (
    id_venta VARCHAR(10) PRIMARY KEY,
    tipo_comprobante VARCHAR(20) NOT NULL,
    num_comprobante VARCHAR(20) NOT NULL,
    fecha DATE NOT NULL,
    hora TIME NOT NULL,
    tipo_documento VARCHAR(10) NOT NULL,
    num_documento VARCHAR(15) NOT NULL,
    cliente_nombre VARCHAR(200) NOT NULL,
    cliente_direccion VARCHAR(150),
    total_gravado DECIMAL(10,2) NOT NULL,
    igv DECIMAL(10,2) NOT NULL,
    total_pagar DECIMAL(10,2) NOT NULL,
    metodo_pago VARCHAR(30) NOT NULL,
    estado VARCHAR(30) DEFAULT 'COMPLETADO',
    id_usuario INT NOT NULL,
    FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario)
);


CREATE TABLE produccion (
    id_produccion VARCHAR(10) NOT NULL,
    id_producto VARCHAR(10) NOT NULL,
    fecha_produccion DATE,
    cantidad_fabricada INT,
    costo_total DECIMAL(10,2),
    id_usuario INT,
    PRIMARY KEY (id_produccion, id_producto),
    FOREIGN KEY (id_producto) REFERENCES producto(id_producto),
    FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario)
);

CREATE TABLE cotizacion_detalle (
    id_cotizacion VARCHAR(10) NOT NULL,
    id_materia_prima VARCHAR(10) NOT NULL,
    cantidad_sol DECIMAL(10,2) NOT NULL,
    precio_propi DECIMAL(10,2) NOT NULL,
    PRIMARY KEY (id_cotizacion, id_materia_prima),
    FOREIGN KEY (id_materia_prima) REFERENCES materia_prima(id_materia_prima)
);

CREATE TABLE compra (
    id_compra VARCHAR(10) PRIMARY KEY,
    id_cotizacion VARCHAR(10) NOT NULL,
    id_proveedor VARCHAR(10) NOT NULL,
    fecha_compra DATE NOT NULL,
    id_usuario INT NOT NULL,
    total DECIMAL(10,2) NOT NULL,
    FOREIGN KEY (id_cotizacion, id_proveedor) REFERENCES cotizacion(id_cotizacion, id_proveedor),
    FOREIGN KEY (id_usuario) REFERENCES usuario(id_usuario)
);

CREATE TABLE venta_producto (
    id_venta VARCHAR(10) NOT NULL,
    id_producto VARCHAR(10) NOT NULL,
    cantidad INT NOT NULL,
    precio_unitario DECIMAL(10,2) NOT NULL,
    subtotal DECIMAL(10,2) NOT NULL,
    PRIMARY KEY (id_venta, id_producto), 
    FOREIGN KEY (id_venta) REFERENCES venta(id_venta) ON DELETE CASCADE,
    FOREIGN KEY (id_producto) REFERENCES producto(id_producto)
);

CREATE TABLE compra_detalle (
    id_compra VARCHAR(10) NOT NULL,
    id_materia_prima VARCHAR(10) NOT NULL,
    cantidad_comprada DECIMAL(10,2) NOT NULL,
    precio_final DECIMAL(10,2) NOT NULL,
    PRIMARY KEY (id_compra, id_materia_prima),
    FOREIGN KEY (id_compra) REFERENCES compra(id_compra),
    FOREIGN KEY (id_materia_prima) REFERENCES materia_prima(id_materia_prima)
);

CREATE PROCEDURE LOGIN(
    IN cor VARCHAR(100),
    IN con VARCHAR(255),
    IN tok INT
)
BEGIN
    DECLARE v_estado VARCHAR(20);
    SELECT estado INTO v_estado FROM usuario WHERE correo = cor AND contrasena = con;
    
    IF v_estado = 'CONTRATADO' THEN
        UPDATE usuario SET token = tok WHERE correo = cor AND contrasena = con;
        SELECT rol FROM usuario WHERE correo = cor AND contrasena = con;
    ELSE
        SELECT NULL;
    END IF;
END$$

DROP PROCEDURE IF EXISTS registrar_materia_prima;
DELIMITER $$
CREATE PROCEDURE registrar_materia_prima(
    IN op CHAR(1),
    IN id VARCHAR(10),
    IN nom VARCHAR(100),
    IN uni VARCHAR(30),
    IN stock DECIMAL(10,2),
    IN precio DECIMAL(10,2),
    IN stock_min DECIMAL(10,2)
)
BEGIN
    DECLARE v INT;
    DECLARE cod VARCHAR(10);
    
    IF op = 'I' THEN
        SELECT IFNULL(MAX(id_materia_prima), 'MP0000') INTO cod FROM materia_prima;
        SET v = SUBSTRING(cod, 3) + 1;
        SET cod = CONCAT('MP', LPAD(v, 4, '0'));
        INSERT INTO materia_prima (id_materia_prima, nombre_mp, unidad_mp, stock_mp, precio_unitario, stock_minimo)
        VALUES (cod, nom, uni, stock, precio, stock_min);
    ELSEIF op = 'U' THEN
        UPDATE materia_prima
        SET nombre_mp = nom,
            unidad_mp = uni,
            stock_mp = stock,
            precio_unitario = precio,
            stock_minimo = stock_min
        WHERE id_materia_prima = id;
    ELSEIF op = 'D' THEN
        DELETE FROM materia_prima WHERE id_materia_prima = id;
    END IF;
END$$
DELIMITER ;

DROP PROCEDURE IF EXISTS registrar_producto_completo;
DELIMITER $$
CREATE PROCEDURE registrar_producto_completo(
    IN nom VARCHAR(100),
    IN precio DECIMAL(10,2),
    IN stock_min INT,
    IN materiales_json TEXT
)
BEGIN
    DECLARE id_rec VARCHAR(10);
    DECLARE id_prd VARCHAR(10);
    DECLARE v INT;
    DECLARE i INT DEFAULT 0;
    DECLARE total INT;
    DECLARE id_mp VARCHAR(10);
    DECLARE cant DECIMAL(10,2);
    DECLARE obtener_monto DECIMAL(10,2) DEFAULT 0.00;
    DECLARE costo_pro DECIMAL(10,2) DEFAULT 0.00;

    SELECT IFNULL(MAX(id_receta), 'RC0000') INTO id_rec FROM receta;
    SET v = SUBSTRING(id_rec, 3) + 1;
    SET id_rec = CONCAT('RC', LPAD(v, 4, '0'));
    
    INSERT INTO receta (id_receta, nom_receta, costo_produccion)
    VALUES (id_rec, CONCAT('Receta-', nom), 0.00);
    
    SELECT IFNULL(MAX(id_producto), 'PRD0000') INTO id_prd FROM producto;
    SET v = SUBSTRING(id_prd, 4) + 1;
    SET id_prd = CONCAT('PRD', LPAD(v, 4, '0'));

    INSERT INTO producto (id_producto, nom_producto, stock_prd, precio_prd, id_receta, stock_minimo)
    VALUES (id_prd, nom, 0, precio, id_rec, stock_min);
    
    SET total = JSON_LENGTH(materiales_json);
    WHILE i < total DO
        SET id_mp = JSON_UNQUOTE(JSON_EXTRACT(materiales_json, CONCAT('$[', i, '].idMateriaPrima')));
        SET cant = JSON_EXTRACT(materiales_json, CONCAT('$[', i, '].cantidad'));

        INSERT INTO receta_mp (id_receta, id_materia_prima, cantidad)
        VALUES (id_rec, id_mp, cant);
        
        SET obtener_monto = (SELECT precio_unitario FROM materia_prima WHERE id_materia_prima = id_mp) * cant;
        SET costo_pro = costo_pro + obtener_monto;
        SET i = i + 1;
    END WHILE;
    
    UPDATE receta
    SET costo_produccion = costo_pro
    WHERE id_receta = id_rec;
END$$
DELIMITER ;

DROP PROCEDURE IF EXISTS validar_stock_mp;
  DELIMITER $$
  CREATE PROCEDURE validar_stock_mp(
      IN p_id_producto VARCHAR(10),
      IN p_cantidad INT
  )
  BEGIN
      DECLARE v_nombre VARCHAR(100);

      SELECT mp.nombre_mp INTO v_nombre
      FROM materia_prima mp
      INNER JOIN receta_mp rmp ON mp.id_materia_prima = rmp.id_materia_prima
      INNER JOIN producto p ON rmp.id_receta = p.id_receta
      WHERE p.id_producto = p_id_producto
        AND (mp.stock_mp - (rmp.cantidad * p_cantidad)) < 0
      LIMIT 1;

      IF v_nombre IS NULL THEN
          SELECT 'OK' AS resultado;
      ELSE
          SELECT CONCAT('Sin stock: ', v_nombre) AS resultado;
      END IF;
  END$$
  DELIMITER ;

DROP PROCEDURE IF EXISTS registrar_tanda_produccion;
DELIMITER $$
CREATE PROCEDURE registrar_tanda_produccion(
    IN p_id_usuario INT,
    IN p_productos_json TEXT
)
BEGIN
    DECLARE v_id_produccion VARCHAR(10);
    DECLARE v_cant_productos INT;
    DECLARE i INT DEFAULT 0;
    DECLARE v_id_producto VARCHAR(10);
    DECLARE v_cantidad INT;
    DECLARE v_costo DECIMAL(10,2);
    
    DECLARE EXIT HANDLER FOR SQLEXCEPTION
    BEGIN
        ROLLBACK;
        RESIGNAL;
    END;
    
    START TRANSACTION;
    
    SELECT IFNULL(MAX(id_produccion), 'TD0000') INTO v_id_produccion FROM produccion;
    SET v_id_produccion = CONCAT('TD', LPAD(SUBSTRING(v_id_produccion, 3) + 1, 4, '0'));
    
    SET v_cant_productos = JSON_LENGTH(p_productos_json);
    WHILE i < v_cant_productos DO
        SET v_id_producto = JSON_UNQUOTE(JSON_EXTRACT(p_productos_json, CONCAT('$[', i, '].idProducto')));
        SET v_cantidad = CAST(JSON_UNQUOTE(JSON_EXTRACT(p_productos_json, CONCAT('$[', i, '].cantidadFabricada'))) AS SIGNED);
        SET v_costo = CAST(JSON_UNQUOTE(JSON_EXTRACT(p_productos_json, CONCAT('$[', i, '].costoTotal'))) AS DECIMAL(10,2));
        
        INSERT INTO produccion (id_produccion, id_producto, fecha_produccion, cantidad_fabricada, costo_total, id_usuario)
        VALUES (v_id_produccion, v_id_producto, CURDATE(), v_cantidad, v_costo, p_id_usuario);
        
        UPDATE producto 
        SET stock_prd = stock_prd + v_cantidad 
        WHERE id_producto = v_id_producto;
        
        UPDATE materia_prima mp
        INNER JOIN receta_mp rmp ON mp.id_materia_prima = rmp.id_materia_prima
        INNER JOIN producto p ON rmp.id_receta = p.id_receta
        SET mp.stock_mp = mp.stock_mp - (rmp.cantidad * v_cantidad)
        WHERE p.id_producto = v_id_producto;
        
        SET i = i + 1;
    END WHILE;
    COMMIT;
END$$
DELIMITER ;

DROP PROCEDURE IF EXISTS administrar_proveedor;
DELIMITER $$
CREATE PROCEDURE administrar_proveedor(
    IN p_op CHAR(1),
    IN p_id_proveedor VARCHAR(10),
    IN p_nombre_proveedor VARCHAR(100),
    IN p_ruc VARCHAR(20),
    IN p_telefono VARCHAR(20),
    IN p_direccion VARCHAR(150),
    IN p_correo VARCHAR(100),
    OUT p_mensaje VARCHAR(100)
)
BEGIN
    DECLARE v_id VARCHAR(10);
    
    IF p_op = 'I' THEN
        IF EXISTS (SELECT 1 FROM proveedor WHERE ruc = p_ruc) THEN
            SET p_mensaje = 'El RUC ya existe';
        ELSE
            SELECT IFNULL(MAX(id_proveedor), 'PV0000') INTO v_id FROM proveedor;
            SET v_id = CONCAT('PV', LPAD(SUBSTRING(v_id, 3) + 1, 4, '0'));
            INSERT INTO proveedor (id_proveedor, nombre_proveedor, ruc, telefono, direccion, correo)
            VALUES (v_id, p_nombre_proveedor, p_ruc, p_telefono, p_direccion, p_correo);
            SET p_mensaje = 'OK'; 
        END IF;
    ELSEIF p_op = 'U' THEN
        IF EXISTS (SELECT 1 FROM proveedor WHERE ruc = p_ruc AND id_proveedor <> p_id_proveedor) THEN
            SET p_mensaje = 'El RUC ya existe';
        ELSE
            UPDATE proveedor 
            SET nombre_proveedor = p_nombre_proveedor,
                ruc = p_ruc,
                telefono = p_telefono,
                direccion = p_direccion,
                correo = p_correo
            WHERE id_proveedor = p_id_proveedor;
            SET p_mensaje = 'OK';
        END IF;
    END IF;
END$$
DELIMITER ;

DROP PROCEDURE IF EXISTS registrar_cotizacion;
DELIMITER $$
CREATE PROCEDURE registrar_cotizacion(
    IN p_id_usuario INT,
    IN p_materiales_json TEXT,
    IN p_proveedores_json TEXT,
    OUT p_id_cotizacion VARCHAR(10)
)
BEGIN
    DECLARE v_id_cot VARCHAR(10);
    DECLARE v_total INT;
    DECLARE i INT DEFAULT 0;
    DECLARE v_id_mp VARCHAR(10);
    DECLARE v_cantidad DECIMAL(10,2);
    DECLARE v_precio DECIMAL(10,2);
    DECLARE v_id_prov VARCHAR(10);
    DECLARE v_costo_total DECIMAL(10,2);
    
    SELECT IFNULL(MAX(id_cotizacion), 'COT0000') INTO v_id_cot FROM cotizacion;
    SET v_id_cot = CONCAT('COT', LPAD(SUBSTRING(v_id_cot, 4) + 1, 4, '0'));
    
    SET v_total = JSON_LENGTH(p_materiales_json);
    WHILE i < v_total DO
        SET v_id_mp = JSON_UNQUOTE(JSON_EXTRACT(p_materiales_json, CONCAT('$[', i, '].id_materia_prima')));
        SET v_cantidad = JSON_EXTRACT(p_materiales_json, CONCAT('$[', i, '].cantidad_sol'));
        SET v_precio = JSON_EXTRACT(p_materiales_json, CONCAT('$[', i, '].precio_propi'));
        
        INSERT INTO cotizacion_detalle (id_cotizacion, id_materia_prima, cantidad_sol, precio_propi)
        VALUES (v_id_cot, v_id_mp, v_cantidad, v_precio);
        
        SET i = i + 1;
    END WHILE;

    SELECT SUM(precio_propi) INTO v_costo_total FROM cotizacion_detalle WHERE id_cotizacion = v_id_cot;
    
    SET i = 0;
    SET v_total = JSON_LENGTH(p_proveedores_json);
    WHILE i < v_total DO
        SET v_id_prov = JSON_UNQUOTE(JSON_EXTRACT(p_proveedores_json, CONCAT('$[', i, ']')));
        INSERT INTO cotizacion (id_cotizacion, id_proveedor, fecha_solicitud, id_usuario, costo_total)
        VALUES (v_id_cot, v_id_prov, CURDATE(), p_id_usuario, v_costo_total);
        SET i = i + 1;
    END WHILE;
    
    SET p_id_cotizacion = v_id_cot;
END$$
DELIMITER ;

DROP PROCEDURE IF EXISTS registrar_compra;
DELIMITER $$
CREATE PROCEDURE registrar_compra(
    IN p_id_cotizacion VARCHAR(10),
    IN p_id_proveedor VARCHAR(10),
    IN p_id_usuario INT,
    IN p_detalle_json TEXT,
    OUT p_id_compra VARCHAR(10)
)
BEGIN
    DECLARE v_id_compra VARCHAR(10);
    DECLARE v_total DECIMAL(10,2) DEFAULT 0.00;
    DECLARE v_total_items INT;
    DECLARE i INT DEFAULT 0;
    DECLARE v_id_mp VARCHAR(10);
    DECLARE v_cantidad DECIMAL(10,2);
    DECLARE v_precio DECIMAL(10,2);

    SELECT IFNULL(MAX(id_compra), 'CM0000') INTO v_id_compra FROM compra;
    SET v_id_compra = CONCAT('CM', LPAD(SUBSTRING(v_id_compra, 3) + 1, 4, '0'));
    
    SET v_total_items = JSON_LENGTH(p_detalle_json);
    WHILE i < v_total_items DO
        SET v_cantidad = JSON_EXTRACT(p_detalle_json, CONCAT('$[', i, '].cantidad'));
        SET v_precio = JSON_EXTRACT(p_detalle_json, CONCAT('$[', i, '].precioFinal'));
        SET v_total = v_total + (v_cantidad * v_precio);
        SET i = i + 1;
    END WHILE;

    INSERT INTO compra (id_compra, id_cotizacion, id_proveedor, fecha_compra, id_usuario, total)
    VALUES (v_id_compra, p_id_cotizacion, p_id_proveedor, CURDATE(), p_id_usuario, v_total);
    
    SET i = 0;
    WHILE i < v_total_items DO
        SET v_id_mp = JSON_UNQUOTE(JSON_EXTRACT(p_detalle_json, CONCAT('$[', i, '].idMateriaPrima')));
        SET v_cantidad = JSON_EXTRACT(p_detalle_json, CONCAT('$[', i, '].cantidad'));
        SET v_precio = JSON_EXTRACT(p_detalle_json, CONCAT('$[', i, '].precioFinal'));
        
        INSERT INTO compra_detalle (id_compra, id_materia_prima, cantidad_comprada, precio_final)
        VALUES (v_id_compra, v_id_mp, v_cantidad, v_precio);
        
        UPDATE materia_prima
        SET stock_mp = stock_mp + (v_cantidad * stock_minimo)
        WHERE id_materia_prima = v_id_mp;

        SET i = i + 1;
    END WHILE;

    UPDATE cotizacion SET estado = 'ACEPTADO'
    WHERE id_cotizacion = p_id_cotizacion AND id_proveedor = p_id_proveedor;
    
    SET p_id_compra = v_id_compra;
END$$
DELIMITER ;

DROP PROCEDURE IF EXISTS registrar_venta;
DELIMITER $$
CREATE PROCEDURE registrar_venta(
    IN p_id_usuario        INT,
    IN p_tipo_comprobante  VARCHAR(20),
    IN p_tipo_documento    VARCHAR(10),
    IN p_num_documento     VARCHAR(15),
    IN p_cliente_nombre    VARCHAR(200),
    IN p_cliente_direccion VARCHAR(150),
    IN p_total_gravado     DECIMAL(10,2),
    IN p_igv               DECIMAL(10,2),
    IN p_total_pagar       DECIMAL(10,2),
    IN p_metodo_pago       VARCHAR(30),
    IN p_detalle_json      TEXT,
    OUT p_id_venta         VARCHAR(10)
)
BEGIN
    DECLARE v_id_venta        VARCHAR(10);
    DECLARE v_num_comprobante VARCHAR(20);
    DECLARE v_prefijo         VARCHAR(5);
    DECLARE v_total           INT;
    DECLARE i                 INT DEFAULT 0;
    DECLARE v_id_producto     VARCHAR(10);
    DECLARE v_cantidad        INT;
    DECLARE v_precio          DECIMAL(10,2);
    DECLARE v_subtotal        DECIMAL(10,2);
    
    DECLARE EXIT HANDLER FOR SQLEXCEPTION
    BEGIN
        ROLLBACK;
        RESIGNAL;
    END;

    START TRANSACTION;

    SELECT IFNULL(MAX(id_venta), 'VT0000') INTO v_id_venta FROM venta;
    SET v_id_venta = CONCAT('VT', LPAD(SUBSTRING(v_id_venta, 3) + 1, 4, '0'));
    
    IF p_tipo_comprobante = 'BOLETA' THEN
        SET v_prefijo = 'B001';
    ELSE
        SET v_prefijo = 'F001';
    END IF;
    
    SELECT CONCAT(v_prefijo, '-', LPAD(COUNT(*) + 1, 6, '0')) INTO v_num_comprobante
    FROM venta WHERE tipo_comprobante = p_tipo_comprobante;
    
    INSERT INTO venta (id_venta, tipo_comprobante, num_comprobante, fecha, hora,
                       tipo_documento, num_documento, cliente_nombre, cliente_direccion,
                       total_gravado, igv, total_pagar, metodo_pago, estado, id_usuario)
    VALUES (v_id_venta, p_tipo_comprobante, v_num_comprobante, CURDATE(), CURTIME(),
            p_tipo_documento, p_num_documento, p_cliente_nombre, p_cliente_direccion,
            p_total_gravado, p_igv, p_total_pagar, p_metodo_pago, 'COMPLETADO', p_id_usuario);

    SET v_total = JSON_LENGTH(p_detalle_json);
    WHILE i < v_total DO
        SET v_id_producto = JSON_UNQUOTE(JSON_EXTRACT(p_detalle_json, CONCAT('$[', i, '].idProducto')));
        SET v_cantidad = CAST(JSON_UNQUOTE(JSON_EXTRACT(p_detalle_json, CONCAT('$[', i, '].cantidad'))) AS SIGNED);
        SET v_precio = CAST(JSON_UNQUOTE(JSON_EXTRACT(p_detalle_json, CONCAT('$[', i, '].precioUnitario'))) AS DECIMAL(10,2));
        SET v_subtotal = CAST(JSON_UNQUOTE(JSON_EXTRACT(p_detalle_json, CONCAT('$[', i, '].subtotal'))) AS DECIMAL(10,2));
        
        INSERT INTO venta_producto (id_venta, id_producto, cantidad, precio_unitario, subtotal)
        VALUES (v_id_venta, v_id_producto, v_cantidad, v_precio, v_subtotal);
        
        UPDATE producto
        SET stock_prd = stock_prd - v_cantidad
        WHERE id_producto = v_id_producto;
        
        SET i = i + 1;
    END WHILE;

    COMMIT;
    SET p_id_venta = v_id_venta;
END$$
DELIMITER ;

DROP PROCEDURE IF EXISTS registrar_usuario;
DELIMITER $$
CREATE PROCEDURE registrar_usuario(
    IN p_usuario    VARCHAR(50),
    IN p_rol        VARCHAR(30),
    IN p_telefono   VARCHAR(20),
    IN p_contrasena VARCHAR(255),
    OUT p_mensaje   VARCHAR(100)
)
BEGIN
    DECLARE v_correo VARCHAR(100);
    SET v_correo = CONCAT(p_usuario, '_', LEFT(p_telefono, 3), '_', p_rol, '@kerenmas.pek.com');
    
    IF EXISTS (SELECT 1 FROM usuario WHERE usuario = p_usuario) THEN
        SET p_mensaje = 'El nombre de usuario ya existe';
    ELSEIF EXISTS (SELECT 1 FROM usuario WHERE correo = v_correo) THEN
        SET p_mensaje = 'El correo ya está registrado';
    ELSE
        INSERT INTO usuario (usuario, rol, correo, telefono, contrasena, token, estado)
        VALUES (p_usuario, p_rol, v_correo, p_telefono, p_contrasena, NULL, 'CONTRATADO');
        SET p_mensaje = 'OK';
    END IF;
END$$
DELIMITER ;

DROP PROCEDURE IF EXISTS cambiar_estado_usuario;
DELIMITER $$
CREATE PROCEDURE cambiar_estado_usuario(
    IN p_id_usuario INT,
    OUT p_mensaje VARCHAR(100)
)
BEGIN
    DECLARE v_estado VARCHAR(20);
    SELECT estado INTO v_estado FROM usuario WHERE id_usuario = p_id_usuario;
    
    IF v_estado = 'CONTRATADO' THEN
        UPDATE usuario SET estado = 'DESCONTRATADO', token = NULL WHERE id_usuario = p_id_usuario;
        SET p_mensaje = 'DESCONTRATADO';
    ELSE
        UPDATE usuario SET estado = 'CONTRATADO' WHERE id_usuario = p_id_usuario;
        SET p_mensaje = 'CONTRATADO';
    END IF;
END$$
DELIMITER ;
